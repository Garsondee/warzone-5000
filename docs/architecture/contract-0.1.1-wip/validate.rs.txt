//! `PhysRig::validate()`: the structural and coherence checks every rig must pass before a solver sees it. It returns *every* problem found
//! (never just the first), each naming the station, joint or field at fault. A rig that passes is internally consistent: indices in range,
//! drivetrain wired once, springs agreeing with their preloads, springs carrying the sprung weight, the ground where the wheels say it is.

use std::collections::BTreeSet;

use w5k_math::{scalar, Mat3, Vec3};

use crate::kinematics::inertia_about_axis;
use crate::rig::*;

/// Largest relative disagreement between the vertical preloads and the sprung weight, and between a spring and its preload.
const EQUILIBRIUM_TOLERANCE: f64 = 0.05; // const-ok: a coherence tolerance of the check, not physics
const PRELOAD_TOLERANCE: f64 = 0.02; // const-ok: a coherence tolerance of the check, not physics
/// How far a road wheel's bottom may sit from the belt's ground plane, m.
const GROUND_TOLERANCE_M: f64 = 0.015; // const-ok: a coherence tolerance of the check, not physics

fn finite(x: f64) -> bool {
    x.is_finite()
}

/// A symmetric positive-definite inertia tensor whose diagonal obeys the triangle inequalities (`Ixx <= Iyy + Izz`, ...).
fn physical_inertia(m: &Mat3) -> bool {
    let a = m.m;
    if !m.is_finite() {
        return false;
    }
    let sym = |x: f64, y: f64| (x - y).abs() <= 1e-9 * (1.0 + x.abs() + y.abs()); // const-ok: symmetry tolerance
    if !(sym(a[0][1], a[1][0]) && sym(a[0][2], a[2][0]) && sym(a[1][2], a[2][1])) {
        return false;
    }
    let minor2 = a[0][0] * a[1][1] - a[0][1] * a[1][0];
    a[0][0] > 0.0
        && minor2 > 0.0
        && m.det() > 0.0
        && a[0][0] <= a[1][1] + a[2][2]
        && a[1][1] <= a[0][0] + a[2][2]
        && a[2][2] <= a[0][0] + a[1][1]
}

/// The spring force at compression `c` for the spring kinds that have a closed form here (used to compare a spring with its preload).
fn spring_force_at_rest(k: &SpringKind) -> Option<f64> {
    match k {
        SpringKind::Table { points } => {
            // Interpolate the table at compression 0; extend the end segments linearly.
            if points.len() < 2 {
                return None;
            }
            let seg = points.windows(2).find(|w| w[1].0 >= 0.0).unwrap_or(&points[points.len() - 2..]);
            let (x0, y0, x1, y1) = (seg[0].0, seg[0].1, seg[1].0, seg[1].1);
            let t = (0.0 - x0) / (x1 - x0);
            Some((y0 + (y1 - y0) * t).max(0.0))
        }
        SpringKind::Hydropneumatic { gas_pressure_pa, piston_area_m2, .. } => Some(gas_pressure_pa * piston_area_m2),
        _ => None,
    }
}

fn check_spring(k: &SpringKind, preload_n: f64, what: &str, e: &mut Vec<String>) {
    match k {
        SpringKind::Rigid => {}
        SpringKind::Linear { rate_n_m } => {
            if !(finite(*rate_n_m) && *rate_n_m >= 0.0) {
                e.push(format!("{what}: linear rate must be finite and >= 0"));
            }
        }
        SpringKind::Table { points } => {
            if points.len() < 2
                || points.iter().any(|p| !finite(p.0) || !finite(p.1) || p.1 < 0.0)
                || points.windows(2).any(|w| !(w[1].0 > w[0].0))
            {
                e.push(format!("{what}: a Table spring needs at least two finite points, strictly ascending in compression, with forces >= 0"));
            } else if let Some(f0) = spring_force_at_rest(k) {
                if (f0 - preload_n).abs() > PRELOAD_TOLERANCE * preload_n.abs() + 1.0 {
                    e.push(format!("{what}: the table gives {f0:.0} N at compression 0 but preload_n is {preload_n:.0} N (a Table force is the TOTAL force)"));
                }
            }
        }
        SpringKind::Torsion { rate_nm_rad, arm_length_m, rest_arm_angle_rad } => {
            if !(*rate_nm_rad > 0.0 && *arm_length_m > 0.0 && rest_arm_angle_rad.abs() < 1.5 && finite(*rate_nm_rad)) {
                e.push(format!("{what}: a Torsion spring needs a positive rate, a positive arm length and |rest angle| < 1.5 rad"));
            }
        }
        SpringKind::Hydropneumatic { gas_pressure_pa, gas_volume_m3, piston_area_m2, gamma } => {
            if !(*gas_pressure_pa > 0.0 && *gas_volume_m3 > 0.0 && *piston_area_m2 > 0.0 && (1.0..=2.0).contains(gamma)) {
                e.push(format!("{what}: a Hydropneumatic spring needs positive pressure, volume and area and 1 <= gamma <= 2"));
            } else if let Some(f0) = spring_force_at_rest(k) {
                if (f0 - preload_n).abs() > PRELOAD_TOLERANCE * preload_n.abs() + 1.0 {
                    e.push(format!("{what}: gas pressure times piston area is {f0:.0} N but preload_n is {preload_n:.0} N"));
                }
            }
        }
    }
}

fn check_suspension(s: &SuspensionDef, what: &str, e: &mut Vec<String>) {
    if !(finite(s.preload_n) && s.preload_n >= 0.0) {
        e.push(format!("{what}: preload must be finite and >= 0"));
    }
    check_spring(&s.spring, s.preload_n, what, e);
    let d = &s.damper;
    if ![d.bump_ns_m, d.rebound_ns_m, d.knee_speed_m_s, d.post_knee_ratio, d.friction_n].iter().all(|x| finite(*x) && *x >= 0.0) {
        e.push(format!("{what}: damper coefficients, knee, post-knee ratio and friction must be finite and >= 0"));
    }
    let b = &s.bump_stop;
    if ![b.engage_m, b.rate_n_m, b.progression, b.damping_ns_m].iter().all(|x| finite(*x) && *x >= 0.0) || !(0.0..=1.0).contains(&b.restitution) {
        e.push(format!("{what}: bump stop values must be finite and >= 0 and restitution in 0..1"));
    }
}

fn check_shape(shape: &ProxyShape, what: &str, e: &mut Vec<String>) {
    let ok = match shape {
        ProxyShape::Sphere { radius_m } => *radius_m > 0.0 && finite(*radius_m),
        ProxyShape::Capsule { a, b, radius_m } => *radius_m > 0.0 && a.is_finite() && b.is_finite(),
        ProxyShape::Box { half_m } => half_m.is_finite() && half_m.x > 0.0 && half_m.y > 0.0 && half_m.z > 0.0,
        ProxyShape::Convex { planes } => {
            planes.len() >= 4 && planes.iter().all(|(n, d)| n.is_finite() && (n.length() - 1.0).abs() < 1e-6 && finite(*d)) // const-ok: unit-normal tolerance
        }
    };
    if !ok {
        e.push(format!("{what}: bad collision shape"));
    }
}

fn names_unique<'a>(names: impl Iterator<Item = &'a str>, kind: &str, e: &mut Vec<String>) {
    let mut seen = BTreeSet::new();
    for n in names {
        if !seen.insert(n) {
            e.push(format!("duplicate {kind} name {n:?}"));
        }
    }
}

/// The nearest joint at or above `j` whose drive is a recoil mechanism.
fn recoil_joint_at_or_above(rig: &PhysRig, mut j: Option<usize>) -> Option<usize> {
    while let Some(i) = j {
        if matches!(rig.articulation[i].drive, JointDrive::Recoil(_)) {
            return Some(i);
        }
        j = rig.articulation[i].parent;
    }
    None
}

impl PhysRig {
    /// Structural and coherence checks every rig must pass before a solver sees it. Returns every problem found.
    pub fn validate(&self) -> Result<(), Vec<String>> {
        let mut e: Vec<String> = Vec::new();
        let n = self.stations.len();
        let no = self.drivetrain.outputs.len();

        // ---- the hull and the global numbers
        if !(self.hull.mass_kg > 0.0 && finite(self.hull.mass_kg)) {
            e.push("hull mass must be positive and finite".into());
        }
        if !physical_inertia(&self.hull.inertia_kg_m2) {
            e.push("hull inertia must be a symmetric, positive-definite tensor obeying the triangle inequalities".into());
        }
        if !self.hull.com_m.is_finite() {
            e.push("hull centre of mass must be finite".into());
        }
        if !(self.ride_height_m > 0.0 && finite(self.ride_height_m)) {
            e.push("ride_height_m (the datum's height above the ground at the design pose) must be positive".into());
        }
        if !(1..=MAX_SUBSTEPS).contains(&self.integration.substeps) {
            e.push(format!("substeps must be 1..={MAX_SUBSTEPS} (a rig that needs more is a numerically unstable design)"));
        }
        names_unique(self.stations.iter().map(|s| s.name.as_str()), "station", &mut e);
        names_unique(self.tracks.iter().map(|t| t.name.as_str()), "track", &mut e);
        names_unique(self.articulation.iter().map(|j| j.name.as_str()), "joint", &mut e);
        names_unique(self.muzzles.iter().map(|m| m.name.as_str()), "muzzle", &mut e);
        names_unique(self.linkages.iter().map(|l| l.name.as_str()), "linkage", &mut e);
        names_unique(self.proxies.iter().map(|p| p.name.as_str()), "proxy", &mut e);

        // ---- stations
        let in_linkage: BTreeSet<usize> = self.linkages.iter().flat_map(|l| l.members.iter().map(|m| m.0)).collect();
        for (i, s) in self.stations.iter().enumerate() {
            let w = format!("station {i} ({})", s.name);
            if !(s.wheel.radius_m > 0.0 && finite(s.wheel.radius_m) && s.wheel.width_m > 0.0 && s.wheel.inertia_kg_m2 > 0.0) {
                e.push(format!("{w}: wheel radius, width and inertia must be positive"));
            }
            if !(s.unsprung_mass_kg >= 0.0 && finite(s.unsprung_mass_kg)) {
                e.push(format!("{w}: negative unsprung mass"));
            }
            let rigid = matches!(s.suspension.spring, SpringKind::Rigid);
            if !rigid && s.unsprung_mass_kg <= 0.0 {
                e.push(format!("{w}: a sprung station needs a positive unsprung mass"));
            }
            if rigid && !in_linkage.contains(&i) && s.unsprung_mass_kg > 0.0 {
                e.push(format!("{w}: a rigid station that is not in a linkage is part of the hull: put its mass in hull.mass_kg"));
            }
            if !s.bump_dir.is_finite() || (s.bump_dir.length() - 1.0).abs() > 1e-6 {
                e.push(format!("{w}: bump_dir must be a unit vector"));
            }
            if !(s.bump_travel_m >= 0.0 && s.droop_travel_m >= 0.0 && finite(s.bump_travel_m) && finite(s.droop_travel_m)) {
                e.push(format!("{w}: negative travel"));
            }
            if !s.rest_pos_m.is_finite() {
                e.push(format!("{w}: rest position must be finite"));
            }
            if in_linkage.contains(&i) && !rigid {
                e.push(format!("{w}: a member of a linkage must have SpringKind::Rigid (the spring lives on the linkage)"));
            }
            check_suspension(&s.suspension, &w, &mut e);
            if let Some(o) = s.drive_output {
                match self.drivetrain.outputs.get(o) {
                    None => e.push(format!("{w}: drive_output {o} out of range")),
                    Some(out) if out.station != i => e.push(format!("{w}: drive_output {o} does not point back at this station")),
                    _ => {}
                }
            }
            if s.wheel.kind == WheelKind::Tyre {
                match &s.wheel.tyre {
                    None => e.push(format!("{w}: a Tyre wheel needs a TyreDef")),
                    Some(t) => {
                        if ![t.vertical_stiffness_n_m, t.mu_scale, t.slip_stiffness, t.cornering_stiffness_per_rad, t.relaxation_length_m, t.patch_length_m]
                            .iter()
                            .all(|x| *x > 0.0 && finite(*x))
                        {
                            e.push(format!("{w}: tyre stiffness, grip, slip stiffness, cornering stiffness, relaxation length and patch length must be positive"));
                        }
                        // The free wheel must overlap the ground by a plausible static deflection.
                        let deflection = s.wheel.radius_m - self.ride_height_m - s.rest_pos_m.y;
                        if self.ride_height_m > 0.0 && !(deflection >= 0.0 && deflection <= 0.35 * s.wheel.radius_m) {
                            e.push(format!(
                                "{w}: the free wheel (radius {:.3} m) centred at y = {:.3} m with the datum {:.3} m above the ground overlaps it by {:.3} m; it must overlap by a static deflection (0 .. 0.35 radius)",
                                s.wheel.radius_m, s.rest_pos_m.y, self.ride_height_m, deflection
                            ));
                        }
                    }
                }
            }
            if !s.wheel.patches_x_m.is_empty() && (s.wheel.patches_x_m.iter().any(|x| !finite(*x)) || s.wheel.kind != WheelKind::Tyre) {
                e.push(format!("{w}: patches_x_m must be finite and only a Tyre wheel has contact patches"));
            }
        }

        // ---- linkages and anti-roll bars
        for (i, l) in self.linkages.iter().enumerate() {
            let w = format!("linkage {i} ({})", l.name);
            if l.members.is_empty() || l.members.iter().any(|m| m.0 >= n || !finite(m.1)) {
                e.push(format!("{w}: needs at least one member and every member station must exist"));
            }
            let sum: f64 = l.members.iter().map(|m| m.1).sum();
            if (sum - 1.0).abs() > 1e-6 {
                e.push(format!("{w}: the member weights must sum to 1 (they sum to {sum})"));
            }
            if matches!(l.suspension.spring, SpringKind::Rigid) {
                e.push(format!("{w}: the linkage's own spring must not be Rigid"));
            }
            if matches!(l.rock_limit_m, Some(r) if !(r > 0.0 && finite(r))) {
                e.push(format!("{w}: rock_limit_m must be positive"));
            }
            check_suspension(&l.suspension, &w, &mut e);
        }
        for (i, a) in self.anti_roll.iter().enumerate() {
            if !(a.rate_n_m >= 0.0 && finite(a.rate_n_m)) {
                e.push(format!("anti-roll {i}: rate must be finite and >= 0 (use a linkage for stations that share a spring)"));
            }
            if a.left_station >= n || a.right_station >= n {
                e.push(format!("anti-roll {i}: refers to a missing station"));
            } else if self.stations[a.left_station].side == self.stations[a.right_station].side {
                e.push(format!("anti-roll {i}: the two stations must be on opposite sides"));
            }
        }

        // ---- static equilibrium: the springs carry the sprung weight
        let mut carried = 0.0;
        for (i, s) in self.stations.iter().enumerate() {
            if !matches!(s.suspension.spring, SpringKind::Rigid) && !in_linkage.contains(&i) {
                carried += s.suspension.preload_n * s.bump_dir.y.max(0.0);
            }
        }
        for l in &self.linkages {
            let up: f64 = l.members.iter().filter(|m| m.0 < n).map(|m| m.1 * self.stations[m.0].bump_dir.y.max(0.0)).sum();
            carried += l.suspension.preload_n * up;
        }
        let weight = self.sprung_mass_kg() * scalar::G;
        let sprung_exists = self.stations.iter().any(|s| !matches!(s.suspension.spring, SpringKind::Rigid)) || !self.linkages.is_empty();
        if sprung_exists && (carried - weight).abs() > EQUILIBRIUM_TOLERANCE * weight {
            e.push(format!("static equilibrium: the vertical preloads add up to {carried:.0} N but the sprung weight is {weight:.0} N"));
        }

        // ---- drivetrain
        let dt = &self.drivetrain;
        let eng = &dt.engine;
        if eng.torque_curve.is_empty()
            || eng.torque_curve.iter().any(|p| !finite(p.0) || !finite(p.1) || p.1 < 0.0)
            || eng.torque_curve.windows(2).any(|w| !(w[1].0 > w[0].0))
        {
            e.push("engine: the torque curve must be non-empty, finite, ascending in rpm, with torques >= 0".into());
        }
        if !(eng.idle_rpm >= 0.0 && eng.redline_rpm > eng.idle_rpm && eng.inertia_kg_m2 > 0.0 && finite(eng.redline_rpm)) {
            e.push("engine: idle must be >= 0, below the redline, and the inertia positive".into());
        }
        if !(eng.response_time_s >= 0.0 && eng.idle_fuel_kg_s >= 0.0 && eng.drag_const_nm >= 0.0 && eng.drag_per_rpm_nm >= 0.0) {
            e.push("engine: drag, response time and idle fuel must be >= 0".into());
        }
        let gb = &dt.gearbox;
        if gb.forward_ratios.is_empty() || gb.forward_ratios.iter().chain(&gb.reverse_ratios).any(|r| !(*r > 0.0 && finite(*r))) {
            e.push("gearbox: needs at least one forward ratio, and every ratio must be positive".into());
        }
        if !(gb.efficiency > 0.0 && gb.efficiency <= 1.0) {
            e.push("gearbox: efficiency must be in (0, 1]".into());
        }
        match &dt.coupling {
            CouplingDef::Direct => {}
            CouplingDef::Clutch { max_torque_nm, engage_rpm } => {
                if !(*max_torque_nm > 0.0 && *engage_rpm >= 0.0) {
                    e.push("clutch: capacity must be positive".into());
                }
            }
            CouplingDef::TorqueConverter { stall_ratio, k_factor_rpm_per_sqrt_nm, lockup_speed_ratio } => {
                if !(*stall_ratio >= 1.0 && *k_factor_rpm_per_sqrt_nm > 0.0) || matches!(lockup_speed_ratio, Some(r) if !(*r > 0.0 && *r <= 1.0)) {
                    e.push("torque converter: stall ratio >= 1, a positive K factor and a lock-up speed ratio in (0, 1]".into());
                }
            }
        }
        for (i, o) in dt.outputs.iter().enumerate() {
            if o.station >= n {
                e.push(format!("output {i}: station {} out of range", o.station));
            } else if self.stations[o.station].drive_output != Some(i) {
                e.push(format!("output {i}: station {} does not point back at it (drive_output)", o.station));
            }
            if !(o.final_drive_ratio > 0.0 && o.efficiency > 0.0 && o.efficiency <= 1.0) {
                e.push(format!("output {i}: final drive ratio must be positive and efficiency in (0, 1]"));
            }
        }
        let mut reached = vec![0usize; no];
        let mut leaves = Vec::new();
        collect_outputs(&dt.driveline, &mut leaves);
        for l in leaves {
            if l >= no {
                e.push(format!("driveline refers to output {l}, which does not exist"));
            } else {
                reached[l] += 1;
            }
        }
        if reached.iter().any(|c| *c != 1) {
            e.push("every output must be reached exactly once by the driveline".into());
        }
        self.check_driveline_node(&dt.driveline, &mut e);
        for (i, b) in dt.brakes.iter().enumerate() {
            if b.site == BrakeSite::Wheel && b.station >= n {
                e.push(format!("brake {i}: station {} out of range", b.station));
            }
            if b.site == BrakeSite::Wheel && b.station < n && !self.tracks.is_empty() && self.stations[b.station].drive_output.is_none() {
                e.push(format!("brake {i}: tracked vehicles brake their sprockets; station {} is not driven", b.station));
            }
            if !(b.max_torque_nm >= 0.0 && b.thermal_mass_j_k > 0.0 && b.fade_end_k > b.fade_start_k && (0.0..=1.0).contains(&b.fade_floor)) {
                e.push(format!("brake {i}: torque >= 0, thermal mass > 0, fade_end > fade_start and fade_floor in 0..1"));
            }
            if !(b.reverse_torque_factor > 0.0 && b.reverse_torque_factor <= 1.0 && b.apply_time_s >= 0.0 && b.release_time_s >= 0.0) {
                e.push(format!("brake {i}: reverse factor in (0, 1], lag times >= 0"));
            }
            if b.steering && !(b.service || b.parking) {
                // a steering brake still needs to be a real brake; nothing else to check
            }
        }
        if !dt.modes.is_empty() {
            if dt.default_mode as usize >= dt.modes.len() {
                e.push("drivetrain: default_mode is out of range".into());
            }
            for (i, m) in dt.modes.iter().enumerate() {
                if !(m.ratio_scale > 0.0 && finite(m.ratio_scale))
                    || m.declutched_outputs.iter().any(|o| *o >= no)
                    || m.locked_groups.iter().flatten().any(|o| *o >= no)
                {
                    e.push(format!("drive mode {i} ({}): ratio_scale must be positive and every output index must exist", m.name));
                }
            }
        }

        // ---- tracks
        let mut sides = Vec::new();
        for (ti, t) in self.tracks.iter().enumerate() {
            let w = format!("track {ti} ({})", t.name);
            if t.sprocket >= n || t.idler >= n || t.stations.iter().any(|s| *s >= n) {
                e.push(format!("{w}: a station index is out of range"));
                continue;
            }
            if !t.stations.contains(&t.sprocket) || !t.stations.contains(&t.idler) {
                e.push(format!("{w}: the sprocket and the idler must be in `stations`"));
            }
            if self.stations[t.sprocket].wheel.kind != WheelKind::Sprocket {
                e.push(format!("{w}: `sprocket` is not a Sprocket station"));
            }
            if self.stations[t.idler].wheel.kind != WheelKind::Idler {
                e.push(format!("{w}: `idler` is not an Idler station"));
            }
            if t.stations.first() != Some(&t.sprocket) {
                e.push(format!("{w}: `stations` must be in loop order starting at the sprocket"));
            }
            if t.stations.iter().any(|s| self.stations[*s].side != t.side) {
                e.push(format!("{w}: a listed station is on the other side"));
            }
            if !(t.width_m > 0.0 && t.pitch_m > 0.0 && t.contact_length_m > 0.0 && t.mass_per_m_kg > 0.0 && t.tension_n >= 0.0 && t.samples >= 2) {
                e.push(format!("{w}: width, pitch, contact length, mass per metre must be positive, tension >= 0, at least 2 samples"));
            }
            if t.belt_length_m != 0.0 && t.belt_length_m < t.contact_length_m {
                e.push(format!("{w}: belt_length_m is shorter than the contact length"));
            }
            if ![t.thickness_m, t.resist_c0, t.resist_c1_s_m, t.wheel_contact.vertical_stiffness_n_m, t.wheel_contact.vertical_damping_ns_m].iter().all(|x| *x >= 0.0 && finite(*x)) {
                e.push(format!("{w}: thickness, running resistance and wheel contact values must be finite and >= 0"));
            }
            if matches!(t.shoe_mu_scale_soft, Some(m) if !(m > 0.0)) || !(t.shoe_mu_scale > 0.0) {
                e.push(format!("{w}: shoe friction scales must be positive"));
            }
            // The road wheels stand on the belt, which lies on the ground.
            for s in t.stations.iter().map(|i| &self.stations[*i]).filter(|s| s.wheel.kind == WheelKind::RoadWheel) {
                let bottom = s.rest_pos_m.y - s.wheel.radius_m - t.thickness_m;
                if self.ride_height_m > 0.0 && (bottom - self.ground_y_m()).abs() > GROUND_TOLERANCE_M {
                    e.push(format!(
                        "{w}: road wheel {} bottoms out at y = {bottom:.3} m (with the belt) but the ground plane is at y = {:.3} m",
                        s.name,
                        self.ground_y_m()
                    ));
                }
            }
            sides.push(t.side);
        }
        for side in [Side::Left, Side::Right] {
            if sides.iter().filter(|s| **s == side).count() > 1 {
                e.push(format!("more than one track on side {side:?}"));
            }
        }

        // ---- articulation, muzzles, proxies
        for (i, j) in self.articulation.iter().enumerate() {
            let w = format!("joint {i} ({})", j.name);
            if let Some(p) = j.parent {
                if p >= i {
                    e.push(format!("{w}: parent {p} must precede it"));
                }
            }
            if !j.axis.is_finite() || (j.axis.length() - 1.0).abs() > 1e-6 {
                e.push(format!("{w}: axis must be a unit vector"));
            }
            if !(j.body.mass_kg > 0.0 && finite(j.body.mass_kg)) || !physical_inertia(&j.body.inertia_kg_m2) || !j.body.com_m.is_finite() {
                e.push(format!("{w}: the body needs a positive mass, a finite centre of mass and a physical inertia tensor"));
            }
            if matches!(j.limits, Some((lo, hi)) if !(lo <= hi)) {
                e.push(format!("{w}: limits must be (min, max) with min <= max"));
            }
            match &j.drive {
                JointDrive::Free { friction_si } => {
                    if !(*friction_si >= 0.0 && finite(*friction_si)) {
                        e.push(format!("{w}: friction must be >= 0"));
                    }
                }
                JointDrive::Servo(sv) => {
                    if ![sv.max_rate_si, sv.max_accel_si, sv.max_effort_si, sv.kp_si, sv.kd_si].iter().all(|x| *x > 0.0 && finite(*x)) || !(sv.latency_s >= 0.0 && sv.fallback_rate_si >= 0.0) {
                        e.push(format!("{w}: servo rate, acceleration, effort and gains must be positive; latency and fallback rate >= 0"));
                    } else if i < self.articulation.len() && j.parent.map_or(true, |p| p < i) {
                        let zeros = vec![0.0; self.articulation.len()];
                        let rigid = self.subtree_rigid(i, &zeros);
                        let inertia = match j.kind {
                            JointKind::Revolute => inertia_about_axis(&rigid, j.axis),
                            JointKind::Prismatic => rigid.mass_kg,
                        };
                        if inertia > 0.0 && sv.max_accel_si > sv.max_effort_si / inertia * 1.001 {
                            e.push(format!(
                                "{w}: max_accel_si {:.3} exceeds what max_effort_si allows ({:.3} = effort / subtree inertia {inertia:.0})",
                                sv.max_accel_si,
                                sv.max_effort_si / inertia
                            ));
                        }
                    }
                    if let Some(st) = &sv.stabiliser {
                        if !((0.0..=1.0).contains(&st.rejection) && st.bandwidth_hz > 0.0 && st.latency_s >= 0.0) {
                            e.push(format!("{w}: stabiliser rejection must be in 0..1, bandwidth positive, latency >= 0"));
                        }
                    }
                }
                JointDrive::Recoil(r) => {
                    if j.kind != JointKind::Prismatic {
                        e.push(format!("{w}: a recoil mechanism needs a prismatic joint"));
                    }
                    match j.limits {
                        Some((lo, hi)) if lo == 0.0 && (hi - r.stroke_m).abs() <= 1e-9 * (1.0 + hi.abs()) => {}
                        _ => e.push(format!("{w}: a recoil joint's limits must be (0, stroke_m)")),
                    }
                    if !(r.stroke_m > 0.0 && r.preload_n > 0.0 && r.damper_ns_m >= 0.0 && r.damper_quad_ns2_m2 >= 0.0) {
                        e.push(format!("{w}: recoil stroke and preload must be positive, damping >= 0"));
                    }
                    check_spring(&r.spring, r.preload_n, &w, &mut e);
                }
            }
        }
        for (i, m) in self.muzzles.iter().enumerate() {
            let w = format!("muzzle {i} ({})", m.name);
            if matches!(m.joint, Some(j) if j >= self.articulation.len()) {
                e.push(format!("{w}: joint out of range"));
                continue;
            }
            let wp = &m.weapon;
            if !(wp.trigger < 8 && wp.projectile_mass_kg > 0.0 && wp.muzzle_velocity_m_s > 0.0 && (0.0..=3.0).contains(&wp.recoil_impulse_factor) && wp.dispersion_mrad >= 0.0 && m.caliber_m > 0.0) {
                e.push(format!("{w}: trigger < 8, positive projectile mass, velocity and caliber, recoil factor in 0..3, dispersion >= 0"));
            }
            if let Some(rj) = recoil_joint_at_or_above(self, m.joint) {
                let zeros = vec![0.0; self.articulation.len()];
                let bore = self.body_pose_in_hull(m.joint, &zeros).apply_dir(m.pose.apply_dir(Vec3::new(0.0, 0.0, -1.0)));
                let parent = self.body_pose_in_hull(self.articulation[rj].parent, &zeros);
                let axis = parent.apply_dir(self.articulation[rj].axis);
                if bore.dot(axis) > -0.95 {
                    e.push(format!("{w}: the bore must point opposite to the recoil axis (positive recoil is rearward)"));
                }
            }
        }
        for (i, p) in self.proxies.iter().enumerate() {
            let w = format!("proxy {i} ({})", p.name);
            if p.attached_to.is_some() && p.attached_station.is_some() {
                e.push(format!("{w}: attached_to and attached_station are exclusive"));
            }
            if matches!(p.attached_to, Some(j) if j >= self.articulation.len()) || matches!(p.attached_station, Some(s) if s >= n) {
                e.push(format!("{w}: attachment out of range"));
            }
            check_shape(&p.shape, &w, &mut e);
        }
        self.check_combat(&mut e);

        if e.is_empty() {
            Ok(())
        } else {
            Err(e)
        }
    }

    fn check_driveline_node(&self, node: &DriveNode, e: &mut Vec<String>) {
        match node {
            DriveNode::Output(_) => {}
            DriveNode::Diff { kind, ratio, bias, split, efficiency, children } => {
                if children.is_empty() {
                    e.push("a Diff needs at least one child".into());
                }
                if !(*ratio > 0.0 && finite(*ratio) && *efficiency > 0.0 && *efficiency <= 1.0) {
                    e.push("a Diff's ratio must be positive and its efficiency in (0, 1]".into());
                }
                if *kind == DiffKind::LimitedSlip && !(*bias >= 1.0 && finite(*bias)) {
                    e.push("a LimitedSlip Diff needs a torque-bias ratio >= 1".into());
                }
                if !split.is_empty() && (split.len() != children.len() || split.iter().any(|x| *x < 0.0) || (split.iter().sum::<f64>() - 1.0).abs() > 1e-6) {
                    e.push("a Diff's split must have one non-negative fraction per child, summing to 1".into());
                }
                for c in children {
                    self.check_driveline_node(c, e);
                }
            }
            DriveNode::SteerUnit { ratio, law, children, .. } => {
                if !(*ratio > 0.0 && finite(*ratio)) {
                    e.push("a SteerUnit's ratio must be positive".into());
                }
                if law.detents.iter().any(|d| !(*d > 0.0 && *d <= 1.0)) || law.diff_ratio_by_gear.iter().any(|d| !finite(*d)) {
                    e.push("steering law: detents must be in (0, 1] and ratios finite".into());
                }
                if matches!(law.steer_brakes, Some(b) if b.iter().any(|i| *i >= self.drivetrain.brakes.len())) {
                    e.push("steering law: steer_brakes refers to a missing brake".into());
                }
                let outs: Vec<Option<usize>> = children.iter().map(|c| if let DriveNode::Output(i) = c { Some(*i) } else { None }).collect();
                if outs.len() != 2 || outs.iter().any(|o| o.is_none()) {
                    e.push("a SteerUnit needs exactly two Output children, [left, right]".into());
                } else {
                    let side = |o: usize| self.drivetrain.outputs.get(o).and_then(|out| self.stations.get(out.station)).map(|s| s.side);
                    if side(outs[0].unwrap_or(0)) != Some(Side::Left) || side(outs[1].unwrap_or(0)) != Some(Side::Right) {
                        e.push("a SteerUnit's children must be [left, right] by the sides of the stations they drive".into());
                    }
                }
                for c in children {
                    self.check_driveline_node(c, e);
                }
            }
        }
    }

    fn check_combat(&self, e: &mut Vec<String>) {
        let c = &self.combat;
        let nj = self.articulation.len();
        let body_ok = |b: &Option<usize>| b.map_or(true, |j| j < nj);
        for (i, m) in c.materials.iter().enumerate() {
            if !(m.density_kg_m3 > 0.0 && m.rha_kinetic >= 0.0 && m.rha_chemical >= 0.0) {
                e.push(format!("combat material {i} ({}): density must be positive, protection >= 0", m.name));
            }
        }
        for (i, a) in c.armour.iter().enumerate() {
            let w = format!("armour solid {i} ({})", a.name);
            if !body_ok(&a.body) || a.material >= c.materials.len() || matches!(a.module, Some(m) if m >= c.modules.len()) {
                e.push(format!("{w}: body, material or module out of range"));
            }
            check_shape(&a.shape, &w, e);
        }
        for (i, m) in c.modules.iter().enumerate() {
            let w = format!("module {i} ({})", m.name);
            if !(m.health_j > 0.0) || m.volumes.iter().any(|v| !body_ok(&v.body)) {
                e.push(format!("{w}: health must be positive and every volume's body must exist"));
            }
            for v in &m.volumes {
                check_shape(&v.shape, &w, e);
            }
            for fx in &m.effects {
                use crate::combat::ModuleEffect as F;
                let ok = match fx {
                    F::Derate { target, factor } => (0.0..=1.0).contains(factor) && self.derate_target_ok(target),
                    F::CookOff { chance, .. } => (0.0..=1.0).contains(chance),
                    F::Fire { power_w } => *power_w >= 0.0,
                };
                if !ok {
                    e.push(format!("{w}: a module effect has a factor or chance outside 0..1, a negative power, or a target that does not exist"));
                }
            }
        }
        for (i, m) in c.mass_items.iter().enumerate() {
            if !body_ok(&m.body) || !(m.full_mass_kg >= 0.0) || matches!(m.module, Some(k) if k >= c.modules.len()) {
                e.push(format!("mass item {i} ({}): body, mass or module out of range", m.name));
            }
        }
        for (i, s) in c.sensors.iter().enumerate() {
            if !body_ok(&s.body) || !(s.fov_rad > 0.0) || matches!(s.boresight_to, Some(m) if m >= self.muzzles.len()) {
                e.push(format!("sensor {i} ({}): body, field of view or boresight out of range", s.name));
            }
        }
    }

    fn derate_target_ok(&self, t: &crate::combat::DerateTarget) -> bool {
        use crate::combat::DerateTarget as D;
        match t {
            D::Station(i) => *i < self.stations.len(),
            D::Joint(i) => *i < self.articulation.len(),
            D::Muzzle(i) => *i < self.muzzles.len(),
            D::Sensor(i) => *i < self.combat.sensors.len(),
            _ => true,
        }
    }
}
